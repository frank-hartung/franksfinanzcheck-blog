{{- /* Accessible, build-time chart renderer. Source of truth: data/datasets/.yaml Supported in v1: bar, line; one non-negative series. */ -}} {{- $key := .Get "dataset" -}} {{- if not $key }}{{ errorf "chart: Parameter dataset fehlt in %s" .Page.File.Path }}{{ end -}} {{- /* Kein site.Data/hugo.Data: Der Repo-Datenbaum enthält absichtlich JSONL- Auditspuren. Ein globaler Data-Zugriff würde Hugo zwingen, diese als Konfiguration zu parsen. Deshalb genau die freigegebene YAML-Datei lesen. */ -}} {{- $dataPath := printf "data/datasets/%s.yaml" $key -}} {{- if not (os.FileExists $dataPath) }}{{ errorf "chart: Datensatz %q nicht gefunden (Seite %s)" $key .Page.File.Path }}{{ end -}} {{- $dataset := os.ReadFile $dataPath | transform.Unmarshal (dict "format" "yaml") -}} {{- $kind := $dataset.chart_type -}} {{- if not (in (slice "bar" "line") $kind) }}{{ errorf "chart: %s hat unbekannten chart_type %q" $key $kind }}{{ end -}} {{- $values := $dataset.values -}} {{- $count := len $values -}} {{- if lt $count 2 }}{{ errorf "chart: %s braucht mindestens zwei Werte" $key }}{{ end -}} {{- $max := 0.0 -}} {{- range $values }}{{ if gt (float .value) $max }}{{ $max = float .value }}{{ end }}{{ end -}} {{- if le $max 0.0 }}{{ $max = 1.0 }}{{ end -}} {{- $title := (.Get "title") | default $dataset.title -}} {{- $summary := (.Get "summary") | default $dataset.description -}} {{- $uid := printf "ff-chart-%s-%d" $dataset.id .Ordinal -}} {{- $plotX := 82.0 -}}{{- $plotY := 34.0 -}}{{- $plotW := 626.0 -}}{{- $plotH := 262.0 -}} {{- $baseY := add $plotY $plotH -}}
